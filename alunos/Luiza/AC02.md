import numpy as np
import matplotlib.pyplot as plt

def plot_point(p_orig, p_trans, title):
    plt.figure(figsize=(6, 6))
    plt.scatter(*p_orig, color='blue', label='Original', s=60)
    plt.scatter(*p_trans, color='red', label='Transformado', marker='x', s=100)
    
    # Adicionando anotações de texto para os pontos
    plt.text(p_orig[0], p_orig[1], f' {tuple(np.round(p_orig, 2))}', color='blue')
    plt.text(p_trans[0], p_trans[1], f' {tuple(np.round(p_trans, 2))}', color='red')
    
    plt.axhline(0, color='black', linewidth=0.8)
    plt.axvline(0, color='black', linewidth=0.8)
    plt.grid(True, linestyle='--', alpha=0.7)
    plt.legend()
    plt.title(title)
    plt.axis('equal')
    plt.show()

def plot_shape(s_orig, s_trans, title):
    plt.figure(figsize=(6, 6))
    
    # Fechar os polígonos para plotar a última linha ligando ao primeiro vértice
    s_orig_plot = np.vstack([s_orig, s_orig[0]])
    s_trans_plot = np.vstack([s_trans, s_trans[0]])
    
    plt.plot(s_orig_plot[:,0], s_orig_plot[:,1], 'b-', label='Original', marker='o')
    plt.plot(s_trans_plot[:,0], s_trans_plot[:,1], 'r--', label='Transformado', marker='x')
    
    plt.axhline(0, color='black', linewidth=0.8)
    plt.axvline(0, color='black', linewidth=0.8)
    plt.grid(True, linestyle='--', alpha=0.7)
    plt.legend()
    plt.title(title)
    plt.axis('equal')
    plt.show()

# Ex 1: Translação Simples
p1 = np.array([2, 3])
p1_trans = p1 + np.array([4, -2])
plot_point(p1, p1_trans, "Ex 1: Translação Simples")

# Ex 2: Escala Uniforme
t2 = np.array([[1, 1], [3, 1], [2, 4]])
t2_trans = t2 * 2
plot_shape(t2, t2_trans, "Ex 2: Escala Uniforme")

# Ex 3: Escala Não Uniforme
t3_trans = t2 * np.array([2, 0.5])
plot_shape(t2, t3_trans, "Ex 3: Escala Não Uniforme")

# Ex 4: Rotação na Origem (90° Anti-horário)
p4 = np.array([1, 0])
theta = np.radians(90)
rot_90 = np.array([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]])
p4_trans = p4 @ rot_90.T
plot_point(p4, p4_trans, "Ex 4: Rotação em Torno da Origem (90° Anti-horário)")

# Ex 5: Rotação de um Polígono (45° Horário)
sq5 = np.array([[1, 1], [1, 4], [4, 4], [4, 1]])
theta5 = np.radians(-45)
rot_45_cw = np.array([[np.cos(theta5), -np.sin(theta5)], [np.sin(theta5), np.cos(theta5)]])
sq5_trans = sq5 @ rot_45_cw.T
plot_shape(sq5, sq5_trans, "Ex 5: Rotação de um Polígono (45° Horário)")

# Ex 6: Reflexão Simples (Eixo Y)
p6 = np.array([2, 5])
p6_trans = p6 * np.array([-1, 1])
plot_point(p6, p6_trans, "Ex 6: Reflexão Simples (Eixo Y)")

# Ex 7: Reflexão de um Triângulo (Eixo X)
t7 = np.array([[2, 3], [4, 3], [3, 5]])
t7_trans = t7 * np.array([1, -1])
plot_shape(t7, t7_trans, "Ex 7: Reflexão (Eixo X)")

# Ex 8: Cisalhamento Horizontal (k=2)
p8 = np.array([2, 3])
matriz_cisalhamento = np.array([[1, 2], [0, 1]])
p8_trans = p8 @ matriz_cisalhamento.T
plot_point(p8, p8_trans, "Ex 8: Cisalhamento Horizontal (k=2)")

# Ex 9: Composição de Transformações
p9 = np.array([3, 2])
p9_1 = p9 + np.array([1, -1])          # Translação
p9_2 = p9_1 @ rot_90.T                 # Rotação 90° CCW
p9_3 = p9_2 * 2                        # Escala uniforme
plot_point(p9, p9_3, "Ex 9: Composição de Transformações")

# Ex 10: Combinação de Transformações em uma Figura
r10 = np.array([[1, 1], [5, 1], [5, 3], [1, 3]])
r10_1 = r10 + np.array([-2, 3])                 # Translação
r10_2 = r10_1 * np.array([1.5, 0.5])            # Escala Não Uniforme
r10_3 = r10_2 * np.array([-1, 1])               # Reflexão Y
plot_shape(r10, r10_3, "Ex 10: Combinação de Transformações")
